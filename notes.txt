        elif keyboard.Key.cmd in pressed and key.char == "j":
            vim_down()
        elif keyboard.Key.cmd in pressed and key.char == "k":
            vim_up()

            vim_left()

def vim_down():
    pyautogui.press('down')

def vim_up():
    pyautogui.press('up')
def vim_left():
    pyautogui.press('left')
def vim_right():
    pyautogui.press('right')
